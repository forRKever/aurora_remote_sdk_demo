546  RemoteCameraMaskImage result;
547 
548  slamtec_aurora_sdk_image_desc_t image_desc = {};
549  slamtec_aurora_sdk_camera_mask_image_buffer_t image_buffer = {};
550 
551  // First call to get required buffer size
552  auto err = slamtec_aurora_sdk_camera_mask_get_static_camera_mask_image(_handle, camera_index, &image_desc, &image_buffer, timeout_ms);
553  if (err == SLAMTEC_AURORA_SDK_ERRORCODE_INSUFFICIENT_BUFFER) {
554  // Allocate buffer and try again
555  std::vector<uint8_t> buffer(image_desc.data_size);
556  image_buffer.image_data = buffer.data();
557  image_buffer.image_data_size = image_desc.data_size;
558 
559  err = slamtec_aurora_sdk_camera_mask_get_static_camera_mask_image(_handle, camera_index, &image_desc, &image_buffer, timeout_ms);
560  if (err == SLAMTEC_AURORA_SDK_ERRORCODE_OK) {
561  result = RemoteCameraMaskImage(image_desc, std::move(buffer));
562  if (errcode) *errcode = SLAMTEC_AURORA_SDK_ERRORCODE_OK;
563  return result;
564  }
565  }
566 
567  if (errcode) *errcode = err;
568  return result;
569  }
