107  bool fromLocatorString(const char* input) {
108  std::string protocol;
109  std::string ip;
110  uint16_t tport = SLAMTEC_AURORA_SDK_REMOTE_SERVER_DEFAULT_PORT;
111 
112  std::string inputWrapper = input;
113 
114  size_t protocol_pos = inputWrapper.find('/');
115  if (protocol_pos != std::string::npos) {
116  protocol = inputWrapper.substr(0, protocol_pos);
117  }
118  else {
119  protocol = SLAMTEC_AURORA_SDK_REMOTE_SERVER_DEFAULT_PROTOCOL;
120  protocol_pos = -1;
121  }
122 
123  size_t port_pos = inputWrapper.rfind(':');
124  if (port_pos != std::string::npos) {
125  ip = inputWrapper.substr(protocol_pos + 1, port_pos - protocol_pos - 1);
126  auto&& portStr = inputWrapper.substr(port_pos + 1);
127  tport = (uint16_t)std::stoi(portStr);
128  }
129  else {
130  ip = inputWrapper.substr(protocol_pos + 1);
131  }
132 
133  snprintf(protocol_type, sizeof(protocol_type), "%s", protocol.c_str());
134  snprintf(address, sizeof(address), "%s", ip.c_str());
135  this->port = tport;
136 
137  return true;
138  }

403  case 1: // bgr

409  case 3: // depth (float32)

420  }
